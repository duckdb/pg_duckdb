# ThreadSanitizer suppressions for the test-sanitizers CI job (TSAN_OPTIONS=suppressions=...).
#
# Postgres signal handlers set flags (InterruptPending, ParallelMessagePending, ...) and latches. Postgres is
# single-threaded and accesses those from its main code and its signal handlers without locks, using
# volatile sig_atomic_t variables and memory barriers. pg_duckdb runs Postgres code on DuckDB threads (holding
# the GlobalProcessLock), while the signals are still handled on the main thread, which can't take that lock.
# In Postgres 18 every signal handler runs through wrapper_handler, which matches all of these races.
race:wrapper_handler

# The tests download prebuilt extensions from the DuckDB extension repository. They aren't instrumented, so
# TSan can't see their synchronization, and reports races on memory that they pass between their threads.
race:delta.duckdb_extension
race:vortex.duckdb_extension
race:aws.duckdb_extension
# The AWS SDK doesn't join its event loop cleanup thread before the process exits
thread:aws.duckdb_extension
